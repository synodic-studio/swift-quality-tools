Suggested Fix:
==============

Instead of:
```swift
var body: some View {
    VStack {
        Text("Line 1")
        Text("Line 2")
        Text("Line 3")
        // ... 20+ more lines
    }
}
```

Use computed properties:
```swift
var body: some View {
    VStack {
        headerSection
        contentSection
        footerSection
    }
}

private var headerSection: some View {
    VStack {
        Text("Header")
        // Complex header logic
    }
}

private var contentSection: some View {
    // Complex content logic
}

private var footerSection: some View {
    // Complex footer logic
}
```
